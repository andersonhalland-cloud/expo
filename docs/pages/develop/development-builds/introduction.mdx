import React, { useState } from 'react';
import {
  SafeAreaView, View, Text, FlatList, Pressable, StyleSheet, StatusBar, Alert, ScrollView,
} from 'react-native';
import QRCode from 'react-native-qrcode-svg';

// ---------- Mock data (replace with your backend later) ----------
const VENUES = [
  { id: '1', name: 'The Rail', area: 'Downtown', pass: 15, left: 24, open: '9 PM', perk: 'Skip the line until midnight' },
  { id: '2', name: 'Velvet Room', area: 'Campus Row', pass: 20, left: 8, open: '10 PM', perk: 'No cover + coat check' },
  { id: '3', name: 'Basement 42', area: 'Old Town', pass: 10, left: 40, open: '9 PM', perk: 'Skip the line all night' },
];

const C = {
  bg: '#1E1536', card: '#2B2050', ink: '#F2EEFF', mute: '#9C93C4',
  band: '#FFB400', paper: '#E9E4FF', paperInk: '#1E1536',
};

// In production the server signs this token and the venue scanner verifies it (single use, expires).
const makeToken = (venue) => `CUTLINE:${venue.id}:${Date.now()}:${Math.random().toString(36).slice(2, 10)}`;

export default function App() {
  const [venue, setVenue] = useState(null);
  const [tickets, setTickets] = useState([]);
  const [ticket, setTicket] = useState(null);
  const [tab, setTab] = useState('venues');

  const buy = (v) => {
    // TODO: replace with Stripe PaymentSheet + backend call
    const t = { id: String(Date.now()), venue: v, token: makeToken(v) };
    setTickets((prev) => [t, ...prev]);
    setTicket(t);
    setVenue(null);
  };

  let body;
  if (ticket) body = <TicketScreen t={ticket} onBack={() => setTicket(null)} />;
  else if (venue) body = <VenueScreen v={venue} onBack={() => setVenue(null)} onBuy={buy} />;
  else if (tab === 'tickets') body = <TicketList tickets={tickets} onOpen={setTicket} />;
  else body = <VenueList onOpen={setVenue} />;

  return (
    <SafeAreaView style={s.root}>
      <StatusBar barStyle="light-content" />
      <View style={{ flex: 1 }}>{body}</View>
      {!venue && !ticket && (
        <View style={s.tabs}>
          {['venues', 'tickets'].map((k) => (
            <Pressable key={k} style={s.tab} onPress={() => setTab(k)}>
              <Text style={[s.tabText, tab === k && { color: C.band }]}>
                {k === 'venues' ? 'Tonight' : `My passes (${tickets.length})`}
              </Text>
            </Pressable>
          ))}
        </View>
      )}
    </SafeAreaView>
  );
}

function VenueList({ onOpen }) {
  return (
    <FlatList
      data={VENUES}
      keyExtractor={(v) => v.id}
      contentContainerStyle={{ padding: 20 }}
      ListHeaderComponent={<Text style={s.h1}>Skip the line tonight</Text>}
      renderItem={({ item }) => (
        <Pressable style={s.card} onPress={() => onOpen(item)}>
          <View style={{ flex: 1 }}>
            <Text style={s.venue}>{item.name}</Text>
            <Text style={s.mute}>{item.area} · doors {item.open}</Text>
            <Text style={[s.mute, { marginTop: 6 }]}>{item.left} passes left</Text>
          </View>
          <Text style={s.price}>${item.pass}</Text>
        </Pressable>
      )}
    />
  );
}

function VenueScreen({ v, onBack, onBuy }) {
  return (
    <ScrollView contentContainerStyle={{ padding: 20 }}>
      <Pressable onPress={onBack}><Text style={s.back}>Back</Text></Pressable>
      <Text style={s.h1}>{v.name}</Text>
      <Text style={s.mute}>{v.area} · doors {v.open}</Text>
      <View style={[s.card, { marginTop: 24, flexDirection: 'column', alignItems: 'flex-start' }]}>
        <Text style={s.venue}>Line-skip pass</Text>
        <Text style={[s.mute, { marginTop: 4 }]}>{v.perk}</Text>
        <Text style={[s.price, { marginTop: 12 }]}>${v.pass}</Text>
      </View>
      <Pressable
        style={s.cta}
        onPress={() => Alert.alert('Confirm purchase', `Buy 1 pass to ${v.name} for $${v.pass}?`, [
          { text: 'Cancel', style: 'cancel' },
          { text: 'Buy pass', onPress: () => onBuy(v) },
        ])}
      >
        <Text style={s.ctaText}>Buy pass · ${v.pass}</Text>
      </Pressable>
      <Text style={[s.mute, { marginTop: 12, textAlign: 'center' }]}>Demo mode: no real payment is taken.</Text>
    </ScrollView>
  );
}

function TicketScreen({ t, onBack }) {
  return (
    <View style={{ flex: 1, padding: 20 }}>
      <Pressable onPress={onBack}><Text style={s.back}>Done</Text></Pressable>
      <View style={s.ticket}>
        <Text style={s.ticketVenue}>{t.venue.name}</Text>
        <Text style={{ color: C.paperInk, opacity: 0.7, marginBottom: 24 }}>Line-skip pass · valid tonight</Text>
        <QRCode value={t.token} size={220} backgroundColor={C.paper} color={C.paperInk} />
        <Text style={s.tokenText}>{t.token.slice(-8)}</Text>
        <Text style={{ color: C.paperInk, opacity: 0.7 }}>Show this at the door. One entry per pass.</Text>
      </View>
    </View>
  );
}

function TicketList({ tickets, onOpen }) {
  return (
    <FlatList
      data={tickets}
      keyExtractor={(t) => t.id}
      contentContainerStyle={{ padding: 20 }}
      ListHeaderComponent={<Text style={s.h1}>My passes</Text>}
      ListEmptyComponent={<Text style={s.mute}>No passes yet. Pick a venue under Tonight to buy one.</Text>}
      renderItem={({ item }) => (
        <Pressable style={s.card} onPress={() => onOpen(item)}>
          <Text style={s.venue}>{item.venue.name}</Text>
          <Text style={s.mute}>Tap to show QR</Text>
        </Pressable>
      )}
    />
  );
}

const s = StyleSheet.create({
  root: { flex: 1, backgroundColor: C.bg },
  h1: { color: C.ink, fontSize: 30, fontWeight: '800', marginBottom: 16 },
  card: { backgroundColor: C.card, borderRadius: 16, padding: 18, marginBottom: 12, flexDirection: 'row', alignItems: 'center', justifyContent: 'space-between' },
  venue: { color: C.ink, fontSize: 19, fontWeight: '700' },
  mute: { color: C.mute, fontSize: 14 },
  price: { color: C.band, fontSize: 24, fontWeight: '800' },
  back: { color: C.band, fontSize: 16, marginBottom: 16 },
  cta: { backgroundColor: C.band, borderRadius: 14, padding: 18, marginTop: 24, alignItems: 'center' },
  ctaText: { color: C.paperInk, fontSize: 17, fontWeight: '800' },
  tabs: { flexDirection: 'row', borderTopWidth: 1, borderTopColor: C.card },
  tab: { flex: 1, padding: 16, alignItems: 'center' },
  tabText: { color: C.mute, fontWeight: '700' },
  ticket: { backgroundColor: C.paper, borderRadius: 24, padding: 28, alignItems: 'center', marginTop: 8 },
  ticketVenue: { color: C.paperInk, fontSize: 28, fontWeight: '800', marginBottom: 4 },
  tokenText: { color: C.paperInk, fontSize: 20, fontWeight: '700', letterSpacing: 2, marginVertical: 16 },
});
